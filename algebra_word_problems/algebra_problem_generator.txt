"""
ALGEBRA WORD PROBLEM GENERATOR
==============================
Produces algebra_word_problems.txt: one stored problem per wording. A wording is the
sentence with its numbers blanked. A second draw of the same wording is not stored:
a slider on the unknown uses the stored equation.

Run:   python3 algebra_problem_generator.py [output_file]
Needs: Python 3.9+ only (standard library: random, math, fractions, re, collections).

HOW IT WORKS
- Each problem type is one function decorated with @t (it registers itself in the list T).
- A template picks random numbers, builds the problem text, the equation, the answer, and
  calls a helper that CHECKS the answer satisfies the equation before saving it
  (bad picks are thrown away and retried).
- Helpers (defined in the "SHARED HELPERS" section):
    add(cat, problem, let, equation, steps, answer)  -> store a problem as written
    L(...)    linear equation: asserts the answer works and writes the "Solve" step itself
    Q(...)    quadratic: asserts the root works, writes "ax² + bx + c = 0 → x = ..."
    RAT(...)  equation with the unknown in a denominator
    S2(...)   system of equations (checks every equation)
    INEQ(...) inequality: checks the answer works and the next whole number does not
- The random seed is fixed (20261109), so the same file is produced every run.
  Change the seed in R = random.Random(...) to get a different 1,000.
- Templates are used in turn. A wording already stored (the same sentence with different
  numbers) is skipped. The run stops when a full pass adds nothing new.

WHERE EQUATION NUMBERS COME FROM
Each template shows the formula that turns the problem's numbers into the equation, e.g.
"55% are boys" -> girls = (100 - 55)/100 = 0.45 -> equation 0.45n = girls  (see m_class_percent).
Search for a category name (e.g. "Percent", "Mixture") to find its template(s).
"""
OUT = __import__("sys").argv[1] if len(__import__("sys").argv) > 1 else "algebra_word_problems.txt"

def nm(k=1):
    s=R.sample(NAMES,k); return s if k>1 else s[0]
def co(k): return '' if k==1 else str(k)
def pl(v,unit): return f"{v} {unit if str(v)=='1' else unit+'s'}"
def money(x):

def num(x):
    x=Fr(x)
    if x.denominator==1: return str(x.numerator)
    d=float(x)
    return f"{d:g}" if abs(d*100-round(d*100))<1e-9 else f"{x.numerator}/{x.denominator}"
P=[]
def add(cat,prob,let,eq,steps,ans):
    P.append((cat,prob,let,eq,steps,ans))
T=[]
def t(f): T.append(f); return f

@t
def consecutive():
    k=R.choice([2,3,4]); kind=R.choice(["","even ","odd "]); step=1 if kind=="" else 2
    x=R.randint(5,90)
    if kind=="even " and x%2: x+=1
    if kind=="odd " and x%2==0: x+=1
    S=sum(x+step*i for i in range(k)); assert sum(x+step*i for i in range(k))==S
    terms=" + ".join(["x"]+[f"(x + {step*i})" for i in range(1,k)])
    add("Consecutive integers",f"The sum of {k} consecutive {kind}integers is {S}. What are the integers?",
        "x = the smallest integer",f"{terms} = {S}",
        f"{k}x + {step*k*(k-1)//2} = {S} → {k}x = {S-step*k*(k-1)//2} → x = {x}",
        ", ".join(str(x+step*i) for i in range(k)))

@t
def age_ratio():
    n1,n2=nm(2); k=R.choice([2,3,4]); y=R.randint(3,15); young=R.randint(4,20)
    # parent is k times child now; in y years parent will be m times? use: now k times, y years ago: (k*young - y) = m*(young-y)
    old=k*young
    # in y years sum?
    S=old+young+2*y
    add("Age",f"{n1} is {k} times as old as {n2}. In {y} years, the sum of their ages will be {S}. How old is each now?",
        f"x = {n2}'s age now; {n1} = {k}x",f"(x + {y}) + ({k}x + {y}) = {S}",
        f"{k+1}x + {2*y} = {S} → x = {young}",f"{n2} is {young}, {n1} is {old}")
@t
def age_diff():
    n1,n2=nm(2); d=R.randint(2,30); a=R.randint(5,40); S=2*a+d; y=R.randint(2,10)
    b=a+d
    if R.random()<.5:
        add("Age",f"{n1} is {d} years older than {n2}. The sum of their ages is {S}. How old is each?",
            f"x = {n2}'s age",f"x + (x + {d}) = {S}",f"2x = {S-d} → x = {a}",f"{n2} is {a}, {n1} is {b}")
    else:
        # y years ago, n1 was twice n2: b - y = 2(a - y) -> choose y accordingly
        y=2*a-b
        if y<=0 or a-y<1: return
        add("Age",f"{n1} is {d} years older than {n2}. {y} years ago, {n1} was twice as old as {n2}. How old is each now?",
            f"x = {n2}'s age now",f"(x + {d}) − {y} = 2(x − {y})",f"x + {d-y} = 2x − {2*y} → x = {a}",f"{n2} is {a}, {n1} is {b}")
        assert b-y==2*(a-y)

import re as _re
NAMES2="Ava Ben Carla Dev Elena Finn Grace Hiro Isla Jamal Kira Leo Maya Noah Omar Priya Quinn Rosa Sam Tara Umar Vera Wes Xena Yusuf Zoe Aiden Bea Caleb Dana Emeka Fatima Gus Hana Ivan Jade Kofi Lena Mateo Nia Oscar Pablo Rhea Sofia Theo Uma Victor Wren Yara Zane".split()
def nm(k=1):
    s=R.sample(NAMES2,k); return s if k>1 else s[0]
def F(v):
    v=Fr(v).limit_denominator(10000) if not isinstance(v,Fr) else v
    if v.denominator==1: return str(v.numerator)
    d=float(v)
    if abs(d*10000-round(d*10000))<1e-9: return f"{d:g}"
    return f"{v.numerator}/{v.denominator}"
def py(s):
    s=s.replace('−','-').replace('×','*').replace('²','**2').replace('³','**3').replace('÷','/').replace('·','*').replace('^','**')
    s=_re.sub(r'(?<=[\d\)])\s*(?=[a-z\(])',r'*',s)
    s=_re.sub(r'(?<=[a-z])\s*(?=\()',r'*',s)
    return s
def f_of(eq,var):
    l,r=eq.split('=')
    return lambda v: eval(py(l),{},{var:v})-eval(py(r),{},{var:v})
def sg(b): return f"+ {F(b)}" if b>=0 else f"− {F(-b)}"
def L(cat,prob,let,eq,var,val,ans):
    f=f_of(eq,var); val=Fr(val)
    assert abs(f(float(val)))<1e-7, ("L",eq,val)
    try:
        f0,f1,f2,f3=f(0),f(1),f(2),f(5)
        linear=abs((f2-f1)-(f1-f0))<1e-9 and abs((f3-f0)-5*(f1-f0))<1e-9
    except ZeroDivisionError:
        linear=False
    if not linear:
        add(cat,prob,let,eq,f"Multiply both sides by the denominator and collect terms → {var} = {F(val)}",ans); return
    a=Fr(f1-f0).limit_denominator(10000); b=Fr(f0).limit_denominator(10000)
    if a==0: raise ValueError
    if a<0: a,b=-a,-b
    if a==1: steps=f"Collect like terms → {var} = {F(val)}"
    elif a.numerator==1: steps=f"{var}/{a.denominator} = {F(-b)} → {var} = {F(-b)} × {a.denominator} = {F(val)}"
    elif a==-1: steps=f"−{var} = {F(-b)} → {var} = {F(val)}"
    else:
        A=F(a); A=f"({A})" if "/" in A else A
        steps=f"{A}{var} = {F(-b)} → {var} = {F(val)}"
    add(cat,prob,let,eq,steps,ans)
def Q(cat,prob,let,eq,var,val,ans,other=None):
    f=f_of(eq,var)
    assert abs(f(float(val)))<1e-7, ("Q",eq,val)
    c=Fr(f(0)).limit_denominator(10000); a=Fr((f(1)+f(-1))/2).limit_denominator(10000)-c; b=Fr((f(1)-f(-1))/2).limit_denominator(10000)
    if a<0: a,b,c=-a,-b,-c
    from math import lcm as _l, gcd as _g
    m=_l(a.denominator,b.denominator,c.denominator); a,b,c=a*m,b*m,c*m
    g=_g(int(a),int(b),int(c)) or 1; a,b,c=a/g,b/g,c/g
    disc=b*b-4*a*c
    r=[(-b+s*Fr(math.isqrt(int(disc)) if disc.denominator==1 and math.isqrt(int(disc))**2==disc else math.sqrt(float(disc))).limit_denominator(10000))/(2*a) for s in (1,-1)]
    oth=[x for x in r if abs(float(x)-float(val))>1e-9]
    o=f" (the other root, {F(oth[0])}, is rejected)" if oth else ""
    if other is not None: o=f" ({other})"
    bt=f" {sg(b)}{var}" if b!=0 else ""
    bt=bt.replace(f"+ 1{var}",f"+ {var}").replace(f"− 1{var}",f"− {var}")
    ct=f" {sg(c)}" if c!=0 else ""
    add(cat,prob,let,eq,f"{F(a) if a!=1 else ''}{var}²{bt}{ct} = 0 → {var} = {F(val)}{o}",ans)
def RAT(cat,prob,let,eq,var,val,ans,how="Multiply both sides by the denominators and simplify"):
    f=f_of(eq,var); assert abs(f(float(val)))<1e-7,("R",eq,val)
    add(cat,prob,let,eq,f"{how} → {var} = {F(val)}",ans)
def S2(cat,prob,let,eqs,vals,ans,how="Solve by elimination"):
    for e in eqs:
        l,r=e.split('=')
        assert abs(eval(py(l),{},vals)-eval(py(r),{},vals))<1e-7,("S2",e,vals)
    add(cat,prob,let,";  ".join(eqs),f"{how} → "+", ".join(f"{k} = {F(v)}" for k,v in vals.items()),ans)
def INEQ(cat,prob,let,ineq,var,best,ans,kind):
    # kind 'max' (≤/<) or 'min' (≥/>)
    op=[o for o in ['≤','≥','<','>'] if o in ineq][0]
    l,r=ineq.split(op); pyop={'≤':'<=','≥':'>=','<':'<','>':'>'}[op]
    h=lambda v: eval(f"({py(l)}) {pyop} ({py(r)})",{},{var:v})
    nb=best+1 if kind=='max' else best-1
    assert h(best) and not h(nb), ("INEQ",ineq,best)
    # boundary
    g=lambda v: eval(py(l),{},{var:v})-eval(py(r),{},{var:v})
    a=g(1)-g(0); bnd=Fr(-g(0)/a).limit_denominator(10000)
    word="largest" if kind=='max' else "smallest"
    add(cat,prob,let,ineq,f"{var} {'≤' if (op in '≤<')==(a>0) else '≥'} {F(bnd)} → {var} = {best} ({word} whole number that works)",ans)
def PL(n,unit,plural=None):
    plural=plural or unit+'s'
    return f"{F(n)} {unit if Fr(n)==1 else plural}"
